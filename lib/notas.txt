grafico dos parciais ao logo do tempo
import 'package:flutter/material.dart';
import 'gps_controller.dart';
import 'package:fl_chart/fl_chart.dart';

class PaginaMetros extends StatefulWidget {
  const PaginaMetros({super.key});

  @override
  State<PaginaMetros> createState() => _PaginaMetrosState();
}

class _PaginaMetrosState extends State<PaginaMetros> {
  final GPSController gps = GPSController();

  double _metrosRestantes = 0.0;
  DateTime? _ultimosalto;
  double parcialINmilisseconds = 0.0;

  /// Histórico de parciais
  final List<double> _historicoParciais = [];

  @override
  void initState() {
    super.initState();
    gps.getPermissao().then((_) {
      gps.iniciarTracking();
    });

    gps.addListener(() {
      setState(() {
        _metrosRestantes = _metrosRestantes - gps.distanciaUltimaRemada;
        parcial(gps.distanciaUltimaRemada);
      });
    });
  }

  void parcial(double distanciaUltima) {
    DateTime agora = DateTime.now();
    if (_ultimosalto != null &&
        agora.difference(_ultimosalto!).inMilliseconds > 300) {
      int diferencaTempo = agora.difference(_ultimosalto!).inMilliseconds;

      if (distanciaUltima > 0) {
        parcialINmilisseconds =
            (500 * diferencaTempo) / distanciaUltima;

        // guarda no histórico (máx 30 pontos)
        _historicoParciais.add(parcialINmilisseconds);
        if (_historicoParciais.length > 30) {
          _historicoParciais.removeAt(0);
        }
      }
    }
    _ultimosalto = agora;
  }

  String _formatParcial(double parcialMs) {
    if (parcialMs <= 0) return "--:--";
    int totalSeconds = (parcialMs / 1000).round();
    int minutes = totalSeconds ~/ 60;
    int seconds = totalSeconds % 60;
    return "${minutes.toString().padLeft(1, '0')}:${seconds.toString().padLeft(2, '0')}";
  }

  /// 🔹 Eixo vertical em mm:ss
  Widget _formatAxisLabel(double value, TitleMeta meta) {
    int totalSeconds = (value * 60).round(); // porque value vem em minutos
    int minutes = totalSeconds ~/ 60;
    int seconds = totalSeconds % 60;
    return Text(
      "$minutes:${seconds.toString().padLeft(2, '0')}",
      style: const TextStyle(fontSize: 12),
    );
  }

  Widget _buildGrafico() {
    return SizedBox(
      height: 200,
      child: LineChart(
        LineChartData(
          gridData: FlGridData(show: true),
          titlesData: FlTitlesData(
            bottomTitles: AxisTitles(
              sideTitles: SideTitles(showTitles: false),
            ),
            leftTitles: AxisTitles(
              sideTitles: SideTitles(
                showTitles: true,
                reservedSize: 50,
                getTitlesWidget: _formatAxisLabel, // 🔹 mm:ss no eixo Y
              ),
            ),
          ),
          borderData: FlBorderData(show: true),
          lineBarsData: [
            LineChartBarData(
              spots: _historicoParciais
                  .asMap()
                  .entries
                  .map((e) => FlSpot(
                        e.key.toDouble(),
                        (e.value / 1000) / 60, // minutos
                      ))
                  .toList(),
              isCurved: true,
              dotData: FlDotData(show: false),
              color: Colors.deepPurple,
              barWidth: 3,
            )
          ],
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text("Treino de Remo"),
      ),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text(
              "${gps.distanciaTotal.toStringAsFixed(1)} m",
              style: const TextStyle(
                  fontSize: 32, fontWeight: FontWeight.bold, color: Colors.black),
              textAlign: TextAlign.center,
            ),
            Text(
              "${_metrosRestantes.toStringAsFixed(1)} m restantes",
              style: const TextStyle(
                  fontSize: 20,
                  fontWeight: FontWeight.bold,
                  color: Colors.deepPurple),
              textAlign: TextAlign.center,
            ),
            Text(
              "Parcial: ${_formatParcial(parcialINmilisseconds)} /500m",
              style: const TextStyle(
                  fontSize: 20,
                  fontWeight: FontWeight.bold,
                  color: Colors.deepPurple),
              textAlign: TextAlign.center,
            ),
            const SizedBox(height: 20),
            _buildGrafico(),
          ],
        ),
      ),
    );
  }
}
